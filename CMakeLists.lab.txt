# CMakeLists.lab.txt
# ------------------
# Additive CMake description for the lab branch only.
# This file is NOT included by the upstream build system and must NEVER be
# added to the upstream CMakeLists.txt or any CMakeLists.txt that is already
# present in the v2.0.18 tree.
#
# Purpose: give the pipeline a concrete way to build the vendored cJSON target
# and reference it from lab-only compilation steps (e.g. harnesses that test
# property-string decoding via cJSON).
#
# Usage (from the lab build script or pipeline):
#   cmake -DLAB=ON -C CMakeLists.lab.txt  <source_dir>
# or, more explicitly:
#   cmake -P CMakeLists.lab.txt
#
# The upstream build remains bit-identical to v2.0.18.

cmake_minimum_required(VERSION 3.10)

project(mosquitto_lab LANGUAGES C)

# --------------------------------------------------------------------------- #
# Guard: prevent accidental inclusion in the main upstream build.             #
# --------------------------------------------------------------------------- #
if(DEFINED MOSQUITTO_MAIN_BUILD)
    message(FATAL_ERROR
        "CMakeLists.lab.txt must not be included by the upstream build. "
        "Set -DLAB=ON on a separate lab-only CMake invocation.")
endif()

# --------------------------------------------------------------------------- #
# Vendored cJSON 1.7.14 static library.                                       #
# The library is intentionally kept at a known-vulnerable version for SCA     #
# detection testing. Do not upgrade without updating README.vendor.md.        #
# --------------------------------------------------------------------------- #
set(CJSON_VENDOR_DIR "${CMAKE_CURRENT_LIST_DIR}/.lab/vendor/cjson-1.7.14")

add_library(cjson_vendored STATIC
    "${CJSON_VENDOR_DIR}/cJSON.c"
)

target_include_directories(cjson_vendored PUBLIC
    "${CJSON_VENDOR_DIR}"
)

# Disable warnings as errors for third-party vendored code to avoid
# polluting lab build output with noise from a dependency we do not own.
target_compile_options(cjson_vendored PRIVATE
    -w
)

# --------------------------------------------------------------------------- #
# Lab corpus convenience target — runs generate_corpus.py.                    #
# --------------------------------------------------------------------------- #
find_program(PYTHON3 python3 NAMES python3 python)

if(PYTHON3)
    add_custom_target(lab_corpus
        COMMAND "${PYTHON3}"
            "${CMAKE_CURRENT_LIST_DIR}/.lab/fuzzing/corpus/generate_corpus.py"
        COMMENT "Regenerating fuzzing corpus seeds"
        VERBATIM
    )
endif()

message(STATUS "[lab] cjson_vendored target defined from ${CJSON_VENDOR_DIR}")
message(STATUS "[lab] CMakeLists.lab.txt loaded — upstream build is unmodified.")
